```python
    #    ⚠️ SI 写法必须单列：五个 run 里四个在 cm s^-1 之后再补一行
    #    `k^0≈4.2x10^-4 m s^-1`。只收 10^-2 时这一行无害，但一旦答复**只**给
    #    SI，10^-2 支就整族失配。SIU 用 `(?<!c)m` 挡掉 `cm/s` 自己。
    UNIT = r'cm\s*(?:\\,)?\s*(?:/\s*s|s\s*\^?\{?\s*-\s*1|sec)'
    SIU = r'(?<!c)m\s*(?:\\,)?\s*(?:/\s*s|s\s*\^?\{?\s*-\s*1)'
    MANT = r'4\.(?:1[5-9]|2\d*)'
    K0V = (r'(?<![\d.])' + MANT + r'\s*(?:x\s*10\s*\^?\{?\s*-\s*2|e\s*-\s*0?2)'
           r'|(?<![\d.])0\.04(?:1[5-9]|2\d*)(?![\d])')
    SIV = (r'(?<![\d.])' + MANT + r'\s*(?:x\s*10\s*\^?\{?\s*-\s*4|e\s*-\s*0?4)')
    k0_ok = False
    for pat, unit in ((K0V, UNIT), (SIV, SIU)):
        for m in re.finditer(pat, text, flags=re.IGNORECASE):
            seg = text[max(0, m.start() - 70):m.end() + 60]
            if (re.search(unit, seg, flags=re.IGNORECASE)
                    and re.search(r'k\s*[_^]?\s*\{?\s*0|k0', seg,
                                  flags=re.IGNORECASE)):
                k0_ok = True
                break
        if k0_ok:
            break
    scores["k0_intrinsic_numeric"] = 1.0 if k0_ok else 0.0

    _WEIGHTS = {"k0_intrinsic_numeric": 1.00}
    earned = sum(w for k, w in _WEIGHTS.items() if scores[k] == 1.0)

    # ---- 分支陷阱：命中即扣分（各 0.25），一律在剪掉否证从句的 hyp 上匹配 ----
    #    独立复算的分支指纹（窗口按刚好互斥设计）：
    #      标准 4.17e-2 / 配错 α 与浓度 3.24e-2 / 低频平台当 Rct 2.38e-2(98.0)
    #      或 2.92e-2(80) / 几何面积 1.25e-1 / 零偏置模板 4.47~4.49e-2
    #    ⚠️ 配错 α 这一条不能用符号支 `α_a c_O`：run5 把两项**换序**写成
    #    `α_c c_O^s e^{-α_c Fη/RT}+α_a c_R^s e^{α_a Fη/RT}`，与正确式完全
    #    等价，正则难以稳定区分配对关系。故只用数值指纹。
    scores["hit_swapped_alpha_conc"] = 1.0 if has_any([
        r'(?<![\d.])6\.8[5-9]\d*\s*(?:x\s*10\s*\^?\{?\s*-\s*7|e\s*-\s*0?7)',
        r'(?<![\d.])3\.2[0-5]\d*\s*(?:x\s*10\s*\^?\{?\s*-\s*2|e\s*-\s*0?2)',
        r'(?<![\d.])0\.032[0-5]\d*(?![\d])',
        r'(?<![\d.])3\.[23]\d*\s*(?:x\s*10\s*\^?\{?\s*-\s*4|e\s*-\s*0?4)',
    ], hyp) else 0.0

    #    把低频 Re 平台 98.0 或 98.0-18.0=80 当 Rct（真值 Rct=74.0-18.0=56.0，
    #    低频平台是 Rs+Rct+RD/3）。
    scores["hit_lowfreq_as_rct"] = 1.0 if has_any([
        r'r\s*_?\{?\s*ct[^\n]{0,30}[=≈约]\s*(?:98\.0|98|80\.0|80)(?![\d.])',
        r'(?:98\.0|98\s*-\s*18)[^\n]{0,20}(?:当作|作为|即|视为)[^\n]{0,10}'
        r'r\s*_?\{?\s*ct',
        r'(?<![\d.])2\.9[12]\d*\s*(?:x\s*10\s*\^?\{?\s*-\s*2|e\s*-\s*0?2)',
        r'(?<![\d.])2\.3[89]\d*\s*(?:x\s*10\s*\^?\{?\s*-\s*2|e\s*-\s*0?2)',
        r'(?<![\d.])0\.029[12]\d*(?![\d])', r'(?<![\d.])0\.023[89]\d*(?![\d])',
    ], hyp) else 0.0

    #    漏做面积换算，直接用几何面积 Ageo=0.0710 → k0≈1.25e-1。
    scores["hit_geo_area_k0"] = 1.0 if has_any([
        r'(?<![\d.])1\.2[45]\d*\s*(?:x\s*10\s*\^?\{?\s*-\s*1|e\s*-\s*0?1)',
        r'(?<![\d.])0\.12[456]\d*(?![\d])',
    ], hyp) else 0.0

    #    套零偏置交换电流模板 Rct=RT/(nFj0)、j0=nFk0c → k0≈4.47~4.49e-2。
    scores["hit_zero_bias_template"] = 1.0 if has_any([
        r'(?<![\d.])4\.4[5-9]\d*\s*(?:x\s*10\s*\^?\{?\s*-\s*2|e\s*-\s*0?2)',
        r'(?<![\d.])0\.044[5-9]\d*(?![\d])',
        r'rt\s*/\s*\(?\s*n?\s*f\s*[ij]\s*_?\{?\s*0',
        r'[ij]\s*_?\{?\s*0\s*\}?\s*[=≈]\s*n?\s*f\s*k\s*[_^]?\s*\{?\s*0',
    ], hyp) else 0.0

    penalty = 0.25 * (
        scores["hit_swapped_alpha_conc"] + scores["hit_lowfreq_as_rct"]
        + scores["hit_geo_area_k0"] + scores["hit_zero_bias_template"])

    # ---- 零权重追溯标记：记录谁真做了哪几步派生，不参与计分 ----
    #    为什么只能是追溯而不能计分：这些派生量**五个 run 一个都没写**。
    #    三个正确 run 走的是「以 E-E0′ 为 η + α_a c_R / α_c c_O 直配」的
    #    等价路径，数值上与标准解的局部 Nernst 路径完全一致（括号项都是
    #    5.352e-7），只是从不打印中间量。设成计分项会把三个正确答复全判 0，
    #    那是判「有没有按参考解的写法复述」，不是判对错。过程评价交给
    #    下面的大模型 Rubric（四条 Criterion 恰好覆盖这四步）。
    scores["trace_bv_bracket"] = 1.0 if has_any([
        r'(?<![\d.])5\.35\d*\s*(?:x\s*10\s*\^?\{?\s*-\s*7|e\s*-\s*0?7)',
        r'(?<![\d.])0\.5352?\d*\s*(?:x\s*10\s*\^?\{?\s*-\s*6|e\s*-\s*0?6)',
        r'(?<![\d.])4\.97[0-9]\d*\s*(?:x\s*10\s*\^?\{?\s*-\s*7|e\s*-\s*0?7)',
    ]) else 0.0
    scores["trace_local_nernst"] = 1.0 if has_any([
        r'(?<![\d.])0\.2420\d*', r'(?<![\d.])0\.0332\d*',
        r'(?<![\d.])1\.29[34]\d*(?![\d])']) else 0.0
    scores["trace_ecsa_conductance"] = 1.0 if has_any([
        r'(?<![\d.])11\.9[23]\d*', r'(?<![\d.])0\.0838\d*']) else 0.0
    scores["trace_rd_reflecting"] = 1.0 if has_any([
        r'r\s*_?\{?\s*d\s*\}?[^\n]{0,20}[=≈约]\s*72(?![\d])',
        r'(?<![\d.])24\.0\d*\s*(?:ω|Ω)']) else 0.0

    # 代码侧最终答复证据分：只反映最终答案是否可自动确认。
    scores["auto_final_answer_score"] = round(
        max(0.0, min(1.0, earned - penalty)), 3)

    return scores
```
