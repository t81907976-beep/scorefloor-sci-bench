```python
                  r'[^。\n；;]{0,80}', ' ', text)
    scores["hit_h_phi_048"] = 1.0 if any(
        re.search(p, _rej, re.I) for p in [
            _LAM + r'[^0-9\n]{0,16}0\.4(?:[6-8]\d*|9[0-4]\d*)'
            r'(?![\d])(?!\s*[-+]\s*0?\.?\d)',
        ]) or (scores["humidity_basis_075"] == 0.0
           and has_any([r'\bh\b\s*[=≈约]\s*0\.6'])) else 0.0

    #    陷阱二：端点极限 —— 只取平衡分馏(λ=0)或完全动力(λ=1)，不做联合拟合。
    scores["hit_lambda_endpoint"] = 1.0 if has_any([
        _LAM + r'[^0-9\n]{0,16}(?:1\.00?(?![\d])|0\.00?(?![\d]))',
    ]) else 0.0

    #    陷阱三：误用恒液位、连续补液的开放水库（稳态）模型。
    #    只在同时找不到有限/无补液水库表述时才算踩坑（多数模型提到后予以排除）。
    mentions_finite = has_any([
        r'有限水库', r'无补液', r'不补液', r'瑞利', r'rayleigh',
        r'finite\s+reservoir', r'closed\s+reservoir',
    ])
    scores["hit_open_reservoir"] = 1.0 if ((not mentions_finite) and has_any([
        r'恒液位', r'等量补液', r'开放水库', r'稳态\s*(?:模型|水库)',
        r'steady[-\s]?state\s+reservoir',
    ])) else 0.0

    penalty = 0.25 * (
        scores["hit_h_phi_048"] + scores["hit_lambda_endpoint"]
        + scores["hit_open_reservoir"])

    # 代码侧最终答复证据分：只反映最终答案是否可自动确认。
    scores["auto_final_answer_score"] = round(
        max(0.0, min(1.0, earned - penalty)), 3)

    return scores
```
